-- PASO 2: CREAR TABLA DE CALIFICACIONES
create table public.assignment_submissions_grades (
  id uuid not null default gen_random_uuid (),
  submission_id uuid not null,                  -- Referencia a la entrega
  assignment_id uuid not null,                  -- Para consultas más rápidas
  teacher_id uuid not null,                     -- Profesor que calificó (auth.users)
  score integer null,                           -- Puntuación (0 a X)
  feedback text null,                           -- Comentarios del profesor
  graded_at timestamp with time zone null default now(),
  constraint assignment_submissions_grades_pkey primary key (id),
  constraint assignment_submissions_grades_submission_id_fkey foreign key (submission_id) references assignment_submissions (id) on delete cascade,
  constraint assignment_submissions_grades_assignment_id_fkey foreign key (assignment_id) references assignments (id) on delete cascade,
  constraint assignment_submissions_grades_teacher_id_fkey foreign key (teacher_id) references auth.users (id) on delete cascade,
  constraint assignment_submissions_grades_unique_submission unique (submission_id)  -- Una sola calificación por entrega
) TABLESPACE pg_default;

-- PASO 3: ÍNDICES PARA OPTIMIZAR CONSULTAS
create index idx_assignment_submissions_grades_submission_id on public.assignment_submissions_grades (submission_id);
create index idx_assignment_submissions_grades_teacher_id on public.assignment_submissions_grades (teacher_id);
create index idx_assignment_submissions_grades_assignment_id on public.assignment_submissions_grades (assignment_id);
